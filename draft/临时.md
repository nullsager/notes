local default_deepseek_model = "deepseek-v4-flash"
require("codecompanion").setup({
  adapters = {
    http = {
      deepseek = function()
        return require("codecompanion.adapters").extend("deepseek", {
          env = {
            api_key = "DEEPSEEK_API_KEY",
          },
          schema = {
            model = {
              default = default_deepseek_model,
              choices = {
                ["deepseek-v4-flash"] = {
                  formatted_name = "DeepSeek V4 Flash",
                  -- meta = {
                  --   context_window = 1048576,
                  -- },
                  opts = {
                    can_reason = true,
                    can_use_tools = true,
                  },
                },
                ["deepseek-v4-pro"] = {
                  formatted_name = "DeepSeek V4 Pro",
                  -- meta = {
                  --   context_window = 1048576,
                  -- },
                  opts = {
                    can_reason = true,
                    can_use_tools = true,
                  },
                },
              },
            },

            ["thinking.type"] = {
              default = "enabled",
            },
            reasoning_effort = {
              default = "max",
            },
            -- max_tokens = {
            --   default = 8192,
            -- },
            -- 数字越大，越感性
            temperature = {
              default = 0.4,
            },
          },
        })
      end,
    },
  },
  interactions = {
    chat = {
      adapter = "deepseek",
    },
    inline = {
      adapter = "deepseek",
    },
    cmd = {
      adapter = "deepseek",
    },
  },
  opts = {
    log_level = "INFO",
    language = "Simplified Chinese",
  },
})
-- Open CodeCompanion actions
vim.keymap.set({ "n", "v" }, "<leader>aa", "<cmd>CodeCompanionActions<cr>", {
  noremap = true,
  silent = true,
  desc = "CodeCompanion Actions",
})
-- Toggle chat
vim.keymap.set("n", "<leader>ac", "<cmd>CodeCompanionChat Toggle<cr>", {
  noremap = true,
  silent = true,
  desc = "CodeCompanion Chat Toggle",
})
-- Inline assistant
vim.keymap.set({ "n", "v" }, "<leader>ai", "<cmd>CodeCompanion<cr>", {
  noremap = true,
  silent = true,
  desc = "CodeCompanion Inline",
})
-- Explain selected code
vim.keymap.set("v", "<leader>ae", ":CodeCompanion /explain<cr>", {
  noremap = true,
  silent = true,
  desc = "Explain selected code",
})
local deepseek_models = {
  {
    name = "DeepSeek V4 Flash",
    model = "deepseek-v4-flash",
  },
  {
    name = "DeepSeek V4 Pro",
    model = "deepseek-v4-pro",
  },
}
vim.api.nvim_create_user_command("CCModel", function()
  vim.ui.select(deepseek_models, {
    prompt = "选择 DeepSeek 模型",
    format_item = function(item)
      return item.name
    end,
  }, function(choice)
    if not choice then
      return
    end

    vim.g.codecompanion_deepseek_model = choice.model

    vim.cmd("CodeCompanionChat adapter=deepseek model=" .. choice.model)

    vim.notify("CodeCompanion DeepSeek model: " .. choice.model, vim.log.levels.INFO)
  end)
end, {})

vim.keymap.set("n", "<leader>am", "<cmd>CCModel<cr>", {
  noremap = true,
  silent = true,
  desc = "Switch CodeCompanion DeepSeek Model",
})

vim.api.nvim_create_user_command("CCDeepSeekFlash", function()
  vim.cmd("CodeCompanionChat adapter=deepseek model=deepseek-v4-flash")
end, {})

vim.api.nvim_create_user_command("CCDeepSeekPro", function()
  vim.cmd("CodeCompanionChat adapter=deepseek model=deepseek-v4-pro")
end, {})
